    <script>
        const tg = window.Telegram.WebApp;
        if(tg) { tg.ready(); tg.expand(); }

        // 5 статов в стиле БК для каждого класса
        const heroes = [
            { name: 'Маг', stats: { сила: 5, ловкость: 10, интуиция: 15, выносливость: 12, мудрость: 45 } },
            { name: 'Воин', stats: { сила: 42, ловкость: 12, интуиция: 10, выносливость: 25, мудрость: 3 } },
            { name: 'Стрелок', stats: { сила: 15, ловкость: 40, интуиция: 25, выносливость: 15, мудрость: 5 } },
            { name: 'Танк', stats: { сила: 20, ловкость: 5, интуиция: 5, выносливость: 55, мудрость: 2 } }
        ];
        
        let currentIndex = 0;

        function updateUI() {
            const h = heroes[currentIndex];
            // Меняем цифры в 5 золотых рамках
            document.getElementById('strVal').innerText = h.stats.сила;
            document.getElementById('aglVal').innerText = h.stats.ловкость;
            document.getElementById('intVal').innerText = h.stats.интуиция;
            document.getElementById('vitVal').innerText = h.stats.выносливость;
            document.getElementById('wisVal').innerText = h.stats.мудрость;
        }

        function nextHero() { currentIndex = (currentIndex + 1) % heroes.length; updateUI(); if(tg&&tg.HapticFeedback) tg.HapticFeedback.impactOccurred('light'); }
        function prevHero() { currentIndex = (currentIndex - 1 + heroes.length) % heroes.length; updateUI(); if(tg&&tg.HapticFeedback) tg.HapticFeedback.impactOccurred('light'); }

        function createCharacter() {
            let name = document.getElementById('heroName').value.trim();
            if (name.length < 3) {
                alert("Имя персонажа должно быть не короче 3 символов!");
                return;
            }
            
            if (tg && tg.HapticFeedback) { tg.HapticFeedback.impactOccurred('heavy'); }

            alert(`Персонаж ${name} (${heroes[currentIndex].name}) успешно создан!`);
            window.location.href = "city.html"; 
        }

        updateUI();
    </script>
